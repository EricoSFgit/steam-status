local function updateUI()
    if currentTarget and currentTarget.Character then
        targetLabel.Text = "Alvo: " .. currentTarget.Name
        targetLabel.TextColor3 = Color3.fromRGB(0, 255, 100)
    else
        targetLabel.Text = "Alvo: Nenhum"
        targetLabel.TextColor3 = Color3.new(1, 1, 1)
    end

    if isStuck then
        statusLabel.Text = "Estado: GRUDADO"
        statusLabel.TextColor3 = Color3.fromRGB(0, 255, 100)
    else
        statusLabel.Text = "Estado: Inativo"
        statusLabel.TextColor3 = Color3.fromRGB(255, 200, 0)
    end

    if isSelectingMode then
        selectionStatusLabel.Text = "MODO SELEÇÃO ATIVO - CLIQUE EM UM JOGADOR"
        selectionStatusLabel.Visible = true
        selectBtn.BackgroundColor3 = Color3.fromRGB(255, 150, 0) -- Laranja quando ativo
    else
        selectionStatusLabel.Visible = false
        selectBtn.BackgroundColor3 = Color3.fromRGB(40, 100, 200)
    end
end

local function setMenuState(open)
    menuOpen = open
    -- Animação suave
    local targetTransparency = open and 0.1 or 0.7
    local tweenInfo = TweenInfo.new(0.2)
    TweenService:Create(mainFrame, tweenInfo, {BackgroundTransparency = targetTransparency}):Play()
    
    if open then
        screenGui.Enabled = true
    else
        screenGui.Enabled = true -- Mantém ativo para o highlight funcionar, mas o frame fica transparente
    end
end

local function clearHighlight()
    if currentHighlight then
        currentHighlight:Destroy()
        currentHighlight = nil
    end
end

local function applyHighlightToPart(part)
    clearHighlight()
    if part then
        local highlight = Instance.new("Highlight")
        highlight.FillColor = highlightColor
        highlight.FillTransparency = 0.7 -- Mais visível
        highlight.OutlineColor = highlightColor
        highlight.OutlineTransparency = 0 -- Borda sólida
        highlight.Parent = part
        currentHighlight = highlight
    end
end

local function getTargetFromMouse()
    local mousePos = UserInputService:GetMouseLocation()
    local rayOrigin = workspace.CurrentCamera:ViewportPointToWorldSpace(Vector2.new(mousePos.X, mousePos.Y))
    local rayDir = (workspace.CurrentCamera:ViewportPointToWorldSpaceAtDistance(Vector2.new(mousePos.X, mousePos.Y), 500).Position - rayOrigin).Unit
    
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {character, screenGui}
    params.FilterType = Enum.RaycastFilterType.Exclude
    
    local result = workspace:Raycast(rayOrigin, rayDir * 500, params)
    
    if result and result.Instance then
        local part = result.Instance
        -- Encontra a raiz do personagem
        local model = part:FindFirstAncestorOfClass("Model")
        if model then
            for _, plr in pairs(Players:GetPlayers()) do
                if plr.Character == model then
                    return plr, model:FindFirstChild("HumanoidRootPart")
                end
            end
        end
    end
    return nil, nil
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.F1 then
        setMenuState(not menuOpen)
    end
end)

-- ==========================================
-- LÓGICA DOS BOTÕES
-- ==========================================

selectBtn.MouseButton1Click:Connect(function()
    isSelectingMode = not isSelectingMode
    updateUI()
    
    if isSelectingMode then
        -- Limpa highlight temporário para não confundir
        clearHighlight()
        -- Ativa o highlight dinâmico no mouse
        local mouseHighlightConn
        mouseHighlightConn = RunService.RenderStepped:Connect(function()
            if isSelectingMode then
                local _, hrp = getTargetFromMouse()
                if hrp then
                    applyHighlightToPart(hrp)
                else
                    clearHighlight()
                end
            end
        end)
        
        -- Desativa a conexão quando sair do modo seleção
        task.spawn(function()
            while isSelectingMode do
                task.wait(0.1)
            end
            mouseHighlightConn:Disconnect()
            -- Ao sair, reaplica o highlight no alvo atual se houver
            if currentTarget and targetPart then
                applyHighlightToPart(targetPart)
            end
        end)
    else
        -- Ao desativar, se já tem alvo, mostra o highlight dele
        if currentTarget and targetPart then
            applyHighlightToPart(targetPart)
        else
            clearHighlight()
        end
    end
end)

stickBtn.MouseButton1Click:Connect(function()
    if currentTarget and targetPart then
        isStuck = true
        updateUI()
    else
        statusLabel.Text = "Selecione um alvo primeiro!"
        statusLabel.TextColor3 = Color3.fromRGB(255, 0, 0)
        task.wait(2)
        updateUI()
    end
end)

unstickBtn.MouseButton1Click:Connect(function()
    isStuck = false
    updateUI()
end)

clearBtn.MouseButton1Click:Connect(function()
    currentTarget = nil
    targetPart = nil
    clearHighlight()
    updateUI()
end)

-- ==========================================
-- CLICK PARA SELECIONAR (QUANDO EM MODO SELEÇÃO)
-- ==========================================
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if isSelectingMode and input.UserInputType == Enum.UserInputType.MouseButton1 then
        local player, hrp = getTargetFromMouse()
        if player and player ~= LocalPlayer then
            currentTarget = player
            targetPart = hrp
            updateUI()
            applyHighlightToPart(targetPart)
            -- Opcional: Sair do modo seleção automaticamente após selecionar
            isSelectingMode = false
            updateUI()
        end
    end
end)

-- ==========================================
-- LOOP DE MOVIMENTO (STICK)
-- ==========================================
RunService:BindToRenderStep("StickyFollowLogic", Enum.RenderPriority.Late, function()
    if not isStuck or not targetPart or not humanoidRootPart or not targetPart.Parent then
        return
    end

    -- Verifica se o alvo ainda existe
    if not targetPart:IsDescendantOf(workspace) then
        isStuck = false
        currentTarget = nil
        targetPart = nil
        clearHighlight()
        updateUI()
        return
    end

    local targetPos = targetPart.Position
    local currentPos = humanoidRootPart.Position
    local toTarget = targetPos - currentPos
    
    -- Se estiver muito longe, teleporta
    if toTarget.Magnitude > 40 then
        humanoidRootPart.CFrame = CFrame.new(targetPos - (humanoidRootPart.CFrame.LookVector * stickDistance), targetPos)
        return
    end

    -- Interpolação suave
    local newPos = currentPos:Lerp(toTarget * lerpFactor, lerpFactor)
    local lookTarget = targetPos + Vector3.new(0, 2, 0)
    humanoidRootPart.CFrame = CFrame.lookAt(newPos, lookTarget)
end)

-- ==========================================
-- GERENCIAMENTO DE SPAWN/MORTE
-- ==========================================
local function onCharacterAdded(newChar)
    character = newChar
    humanoidRootPart = character:WaitForChild("HumanoidRootPart")
    humanoid = character:WaitForChild("Humanoid")
    
    isStuck = false
    currentTarget = nil
    targetPart = nil
    clearHighlight()
    updateUI()
end

if character then
    onCharacterAdded(character)
end
LocalPlayer.CharacterAdded:Connect(onCharacterAdded)

-- Inicialização
updateUI()
setMenuState(true)
